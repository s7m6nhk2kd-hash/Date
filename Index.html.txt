  <script>
    const focusSlider = document.getElementById('focusSlider');
    const photoTarget = document.getElementById('photoTarget');
    const crosshair = document.getElementById('crosshair');
    const statusDot = document.getElementById('statusDot');
    const statusText = document.getElementById('statusText');
    const focusVal = document.getElementById('focusVal');

    const OPTIMAL_MIN = 48;
    const OPTIMAL_MAX = 52;
    let audioPlayed = false;

    // Web Audio API Shutter Click Simulation
    function playShutterAudio() {
      try {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!AudioCtx) return;
        const ctx = new AudioCtx();
        
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(120, ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(40, ctx.currentTime + 0.08);
        
        gain.gain.setValueAtTime(0.3, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.08);
        
        osc.connect(gain);
        gain.connect(ctx.destination);
        
        osc.start();
        osc.stop(ctx.currentTime + 0.08);
      } catch (e) {
        // Audio fallback
      }
    }

    function updateFocusState() {
      const val = parseFloat(focusSlider.value);
      focusVal.textContent = Math.round(val) + '%';

      // Distance calculation from optimal 50% focus mark
      const dist = Math.abs(val - 50);
      const blurAmount = (dist * 0.35).toFixed(1);
      
      photoTarget.style.filter = `blur(${blurAmount}px)`;

      if (val >= OPTIMAL_MIN && val <= OPTIMAL_MAX) {
        crosshair.classList.add('sharp');
        statusDot.classList.add('sharp');
        statusText.textContent = "In Focus! Sharp & Ready";
        statusText.style.color = "var(--rose-gold)";
        
        if (!audioPlayed) {
          playShutterAudio();
          audioPlayed = true;
        }
      } else {
        crosshair.classList.remove('sharp');
        statusDot.classList.remove('sharp');
        statusText.textContent = blurAmount > 8 ? "Out of focus" : "Near focus...";
        statusText.style.color = "var(--text-main)";
        audioPlayed = false;
      }
    }

    function setFocus(val) {
      focusSlider.value = val;
      updateFocusState();
    }

    function triggerAutofocus() {
      let current = parseFloat(focusSlider.value);
      const target = 50;
      const step = (target - current) / 15;
      
      let count = 0;
      const interval = setInterval(() => {
        current += step;
        focusSlider.value = current;
        updateFocusState();
        count++;
        if (count >= 15) {
          focusSlider.value = target;
          updateFocusState();
          clearInterval(interval);
        }
      }, 30);
    }

    focusSlider.addEventListener('input', updateFocusState);
    updateFocusState();

    function toggleModal(show) {
      document.getElementById('githubModal').classList.toggle('active', show);
    }
  </script>
